\section{Extraction \& Ingestion Layer}

\begin{frame}[fragile]{NiFi Ingestion Pipeline: Approccio Data-Driven}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \item \textbf{Design Data-Driven}: Pipeline stateless configurata dinamicamente tramite payload JSON iniettato all'avvio.

                \item \textbf{Ingestion \& Validazione}: \texttt{HandleHttpRequest} espone l'endpoint REST per l'avvio da Airflow, mentre \texttt{ValidateJson} esegue un controllo strutturale con schema rigoroso.

                \item \textbf{Propagazione del Dato}: \texttt{EvaluateJsonPath} mappa le chiavi JSON in Attributi del FlowFile, guidando in modo stateless il routing e le decisioni future tramite metadati nativi.

            \end{itemize}
        \end{column}

        \begin{column}{0.45\textwidth}
            \begin{block}{Payload JSON}
                \begin{semiverbatim}\tiny
\{
    "job_id": "FLIGHT_INGEST_run_123",
    "storage_raw": \{
        "type": "HDFS",
        "path": "/data/raw"
    \},
    "storage_preprocessed": \{
        "type": "S3",
        "bucket": "spark-flight-analysis",
        "path": "data/conv"
    \},
    "processing": \{
        "optimization_strategy": "PARTITION_..."
    \},
    "callback": \{
        "url": "http://airflow.../variables/v",
        "method": "PATCH",
        "headers": \{
            "Authorization": "Bearer eyJhbG..."
        \}
    \}
\}
                \end{semiverbatim}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Ingestion, Sicurezza e Gateway}
    \begin{itemize}
        \item \textbf{Estrazione}: Ricezione e spacchettamento degli archivi (\texttt{UnpackContent}).
        \item \textbf{Validazione SHA-1}:
        \begin{itemize}
            \item Controllo di integrità crittografica sui file estratti.
            \item Confronto hash atteso vs calcolato (\texttt{CryptographicHashContent}).
            \item Routing degli errori per isolare dati corrotti.
        \end{itemize}
        \item \textbf{Gateway \& Biforcazione}:
        \begin{itemize}
            \item \textbf{Ramo Raw}: Salvataggio as-is per garanzia del dato originale.
            \item \textbf{Ramo Preprocessing}: Inoltro dei record validati alla trasformazione.
        \end{itemize}
    \end{itemize}

\begin{frame}{Preprocessing e Ottimizzazione in Parquet}
    \begin{itemize}
        \item \textbf{Operatori NiFi per la Trasformazione}:

\section{Computation Layer}

\begin{frame}{Architettura e Astrazione: BaseQuery}
    \begin{itemize}
        \item \textbf{Ecosistema Spark}: Implementazione in Java utilizzando Spark Core (RDD), Spark SQL e MLlib.
        \item \textbf{Design Pattern Template Method}:
        \begin{itemize}
            \item Astrazione tramite la classe \texttt{BaseQuery.java}.
            \item Garanzia di coerenza tra i diversi paradigmi di esecuzione.
        \end{itemize}
        \item \textbf{Benchmarking Comparativo}: Ogni analisi implementa obbligatoriamente tre motori:
        \begin{itemize}
            \item \textbf{RDD API}: Controllo granulare su partizioni e serializzazione.
            \item \textbf{DataFrame API}: Astrazione strutturata (DSL) e ottimizzazione \textit{Catalyst}.
            \item \textbf{Spark SQL}: Interfaccia dichiarativa per logiche di business complesse.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Query 1: Analisi Mensile delle Performance}
    \begin{itemize}
        \item \textbf{Obiettivo}: Ritardi medi, estremi e tasso di cancellazione mensile per vettore.
        \item \textbf{RDD (\texttt{aggregateByKey})}: 
        \begin{itemize}
            \item \texttt{pairs.aggregateByKey(zeroValue, seqOp, combOp)}
            \item Aggregazione locale su partizione per minimizzare il traffico di rete.
        \end{itemize}
        \item \textbf{DataFrame (DSL)}:
        \begin{itemize}
            \item \texttt{flights.groupBy("MONTH", "CARRIER").agg(avg("DELAY"), ...)}
        \end{itemize}
        \item \textbf{Spark SQL}:
        \begin{itemize}
            \item \texttt{SELECT MONTH, AVG(DELAY) FROM flights GROUP BY MONTH, ...}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Query 2: Ranking dei Ritardi in Arrivo}
    \begin{itemize}
        \item \textbf{Obiettivo}: Identificazione delle Top-K tratte più critiche.
        \item \textbf{RDD (Manuale)}:
        \begin{itemize}
            \item \texttt{mapToPair().reduceByKey().takeOrdered(10, comparator)}
        \end{itemize}
        \item \textbf{DataFrame (Window Functions)}:
        \begin{itemize}
            \item \texttt{agg(avg("ARR\_DELAY")).orderBy(col("mean").desc()).limit(10)}
        \end{itemize}
        \item \textbf{Spark SQL}:
        \begin{itemize}
            \item \texttt{SELECT carrier, AVG(ARR\_DELAY) as mean FROM flights ... ORDER BY mean DESC LIMIT 10}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Query 3: Distribuzione e Percentili}
    \begin{itemize}
        \item \textbf{Obiettivo}: Statistiche orarie (Q1, Mediana, Q3).
        \item \textbf{RDD (La sfida)}: \texttt{groupByKey()} su milioni di record causerebbe OOM.
        \item \textbf{DataFrame / SQL (Approssimazione)}:
        \begin{itemize}
            \item \texttt{agg(expr("percentile\_approx(DEP\_DELAY, 0.5)"))}
            \item Uso dell'algoritmo nativo Greenwald-Khanna per stime efficienti.
        \end{itemize}
        \item \textbf{Soluzione RDD Custom}:
        \begin{itemize}
            \item \texttt{PercentileSketch sketch = PercentileSketch.from(config)}
            \item Integrazione di T-Digest e KLL per calcolo distribuito e mergeable.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Gestione Percentili in RDD: Strategie di Sketching}
    \begin{itemize}
        \item \textbf{Design Pattern Strategy}: Implementazione di algoritmi probabilistici in RDD.
        \item \textbf{T-Digest Strategy}:
        \begin{itemize}
            \item Struttura dati basata su centroidi pesati.
            \item Alta precisione nelle "code" della distribuzione (ritardi estremi).
        \end{itemize}
        \item \textbf{KLL Sketch Strategy}:
        \begin{itemize}
            \item Algoritmo stream-based con footprint di memoria fisso e predicibile.
            \item Merge delle strutture dati tra worker senza trasferimento di interi dataset.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Machine Learning: Airline Clustering}
    \begin{itemize}
        \item \textbf{Obiettivo}: Raggruppamento di vettori aerei con profili operativi simili.
        \item \textbf{Modello K-Means} (\textit{Spark MLlib}):
        \begin{itemize}
            \item Inizializzazione tramite \texttt{k-means||} (variante parallela di k-means++).
            \item Convergenza distribuita sui centroidi dei cluster.
        \end{itemize}
        \item \textbf{Data Pipeline}:
        \begin{itemize}
            \item \textbf{VectorAssembler}: Consolidamento delle features analitiche.
            \item \textbf{StandardScaler}: Normalizzazione fondamentale per la distanza Euclidea.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Telemetria Netta: Custom Spark Listener}
    \begin{itemize}
        \item \textbf{Obiettivo}: Misurazione rigorosa dei tempi di esecuzione per il benchmarking.
        \item \textbf{JobTimerListener}:
        \begin{itemize}
            \item Estensione di \texttt{SparkListener} per intercettare gli eventi del DAG Scheduler.
            \item Monitoraggio di \texttt{onJobStart} e \texttt{onJobEnd}.
        \end{itemize}
        \item \textbf{Precisione}: Eliminazione degli overhead di sistema, I/O Driver e task "lazy", garantendo metriche focalizzate puramente sul tempo di calcolo distribuito.
    \end{itemize}
\end{frame}

\section{Valutazione Sperimentale: Metriche}
\begin{frame}{Metodologia di Benchmarking}
    \begin{itemize}
        \item \textbf{Rigore Statistico}: Ogni combinazione (Query $\times$ Backend) è stata eseguita \textbf{5 volte} per garantire la stabilità delle misurazioni.
        \item \textbf{Telemetria}: Utilizzo del \texttt{JobTimerListener} (SparkListener) per isolare il \textit{tempo netto di calcolo} dall'overhead di sistema.
        \item \textbf{Hardware e Infrastruttura}:
        \begin{itemize}
            \item \textbf{EC2 (Self-Hosted)}: Master + Worker su istanze \texttt{t3.small}. Storage basato su HDFS.
            \item \textbf{EMR (Cloud-Native)}: 1 Master + 3 Core su istanze \texttt{m6g.xlarge} (ARM64). Architettura \textit{Shared-Nothing} su Amazon S3 via \texttt{s3a://}.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Confronto API: RDD vs DataFrame vs SQL}
    \begin{itemize}
        \item \textbf{DataFrame/SQL}: Massima stabilità e minori tempi di esecuzione nelle query di aggregazione (es. Q1) grazie all'ottimizzatore \textit{Catalyst} e alla gestione binaria off-heap di \textit{Tungsten}.
        \item \textbf{RDD}: Più performante in task atomici e logiche distribuite custom (Q2 e Q3), nonostante il maggior overhead di serializzazione JVM.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{../Results/charts/telemetry/ec2/part_prun/metrics_ec2_monthly_performance_part_prun_cropped.png}
\end{frame}

\begin{frame}{Ottimizzazione I/O: Pruning vs Pushdown}
    \begin{itemize}
        \item \textbf{Partition Pruning}: Strategia ottimale. Esclude i dati a livello di file system, garantendo un I/O bilanciato e l'assenza di \textit{Workload Skew} sui nodi.
        \item \textbf{Predicate Pushdown}: Richiede la scansione dei metadati Parquet. Genera un volume di lettura superiore e tende a sovraccaricare singoli nodi (colli di bottiglia in fase di scansione).
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{../Results/charts/telemetry/ec2/pred_push/metrics_ec2_monthly_performance_pred_push_cropped.png}
\end{frame}

\begin{frame}{Overhead Infrastrutturale: EC2 vs EMR}
    \begin{itemize}
        \item \textbf{Standalone (EC2)}: Overhead di \textit{Wall Clock Time} significativo dovuto ai tempi di negoziazione delle risorse, bootstrap del cluster e latenze di rete HDFS.
        \item \textbf{Managed (EMR)}: L'integrazione nativa con S3 e la gestione elastica delle istanze contraggono sensibilmente i "tempi morti", massimizzando l'efficienza temporale del job.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{../Results/charts/telemetry/emr/part_prun/metrics_emr_monthly_performance_part_prun_cropped.png}
\end{frame}

\section{Risultati Analitici}

\begin{frame}{Query 3: Scelta dell'Euristica e Approssimazione}
    \begin{itemize}
        \item \textbf{Problema}: Calcolo dei percentili esatti su RDD (\texttt{groupByKey}) causa \textit{Out Of Memory} e colli di bottiglia distribuiti.
        \item \textbf{Soluzione}: Implementazione di algoritmi di \textbf{Quantile Sketching} (T-Digest e KLL).
        \item \textbf{Analisi Comparativa}:
        \begin{itemize}
            \item \textbf{KLL Sketch}: Elevata precisione teorica, ma footprint di memoria superiore.
            \item \textbf{T-Digest}: Ottimizzato per le "code" della distribuzione e più performante.
        \end{itemize}
        \item \textbf{Validazione}: Risultati pressoché identici sul dataset reale; la convergenza dei dati giustifica l'uso di algoritmi probabilistici.
        \item \textbf{Scelta Finale}: \textbf{T-Digest} per il miglior rapporto performance/accuratezza nel dominio aeronautico.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Results/charts/result/Q3_percentiles_trend_AA.png}
\end{frame}

\begin{frame}{Risultati Analitici: Airline Clustering (Q4)}
    \begin{itemize}
        \item \textbf{Model Selection}: Identificazione di \textbf{$K=3$} come numero ottimale di cluster tramite \textit{Silhouette Score}.
        \item \textbf{Profilazione dei Cluster}:
        \begin{itemize}
            \item \textbf{Cluster 0 (Inefficienza)}: \texttt{AA, OH, F9}. Elevata propagazione dei ritardi e instabilità operativa.
            \item \textbf{Cluster 1 (Sensibilità Esterna)}: \texttt{G4}. Forti ritardi meteorologici ma bassissimo tasso di cancellazione.
            \item \textbf{Cluster 2 (Virtuosi)}: \texttt{DL, UA, WN}. Alta capacità di recupero in volo (\textit{Delay Makeup}) e stabilità.
        \end{itemize}
        \item \textbf{Visualizzazione}: Proiezione PCA bidimensionale (perdita di varianza fisiologica rispetto al modello ad alta dimensionalità).
    \end{itemize}
