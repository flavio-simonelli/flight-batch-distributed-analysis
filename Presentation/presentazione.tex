\begin{frame}[fragile, t]{Query 1 - Analisi Mensile Performance: RDD API}

    \queryobjective{Aggregare i voli 2025 di AA e DL su base mensile. Calcolare min, max, media di departure delay solo sui voli non cancellati e il tasso di cancellazione dei voli}

    \begin{rddcodebox}
 flights.mapToPair(v -> ([v.airline, v.month], v))
    .aggregateByKey(
        zeroValue: [sumDelay: 0, maxDelay: -inf, minDelay: +inf, notCancelled: 0, total: 0],
        combiner: (acc, v) -> {
            acc.total += 1
            if !v.cancelled then
                acc.sumDelay     += v.dep_delay
                acc.maxDelay     = MAX(acc.maxDelay, v.dep_delay)
                acc.minDelay     = MIN(acc.minDelay, v.dep_delay)
                acc.notCancelled += 1
            return acc
        },
        reducer: (a1, a2) -> {
            a1.sumDelay     += a2.sumDelay
            a1.maxDelay     = MAX(a1.maxDelay, a2.maxDelay)
            a1.minDelay     = MIN(a1.minDelay, a2.minDelay)
            a1.notCancelled += a2.notCancelled
            a1.total        += a2.total
            return a1
        })
    .map(tuple -> {
        mean = tuple.sumDelay / tuple.notCancelled
        cancRate = (tuple.total - tuple.notCancelled) / tuple.total * 100
        return Row(tuple.month, tuple.airline, media, tuple.minDelay, tuple.maxDelay, cancRate)
    })
    \end{rddcodebox}

\end{frame}

\begin{frame}[fragile, t]{Query 1 - Analisi Mensile Performance: High-Level API}

    \queryobjective{Aggregare i voli 2025 di AA e DL su base mensile. Calcolare min, max, media di departure delay solo sui voli non cancellati e il tasso di cancellazione dei voli}

    \begin{columns}[c]

        \begin{column}{0.53\textwidth}
            \begin{dataframecodebox}
flights
  .groupBy("MONTH", "OP_UNIQUE_CARRIER")
  .agg(
    avg(when(col("CANCELLED") == 0, col("DEP_DELAY"))).as("mean"),
    min(when(col("CANCELLED") == 0, col("DEP_DELAY"))).as("min"),
    max(when(col("CANCELLED") == 0, col("DEP_DELAY"))).as("max"),
    (sum(col("CANCELLED")) / count("*") * 100).as("cancel_rate")
  )
  .withColumnRenamed("MONTH", "month")
  .withColumnRenamed("OP_UNIQUE_CARRIER", "airline")
            \end{dataframecodebox}
        \end{column}

        \begin{column}{0.53\textwidth}
            \begin{sqlcodebox}
SELECT
  MONTH             AS month,
  OP_UNIQUE_CARRIER AS airline,

  AVG(CASE WHEN CANCELLED == 0 THEN DEP_DELAY END) AS 'mean',
  MIN(CASE WHEN CANCELLED == 0 THEN DEP_DELAY END) AS 'min',

\begin{frame}[fragile, t]{Query 3 - Percentili Orari e Min/Max Globali: RDD API}

hourlyRows = validFlights.filter(f -> !f.isNullAt(CRS_DEP_TIME))
    .mapToPair(f -> ( (f.getString(CARRIER), (f.getInt(CRS_DEP_TIME) / 100) % 24), getSafe(f, DEP_DELAY) ))
    .combineByKey(sketch::init, sketch::update, sketch::merge)
    .map(t -> {
        double[] q = sketch.getQuantiles(t.percentile, 0.25, 0.50, 0.75, 0.90)
        return Row(t.key.carrier, t.key.hour, q.p25, q.p50, q.p75, q.p90)
    })
// Pipeline 2: Aggregazione Globale Min/Max per Compagnia
globalRows = validFlights.mapToPair(f -> (f.getString(CARRIER), f))
    .aggregateByKey(
        zeroValue: new double[]{+inf, -inf},  // [0: minDelay, 1: maxDelay]
        combiner: (acc, f) -> {
            double d = getSafe(f, DEP_DELAY)
            acc.min = MIN(acc.min, d)                   acc.max = MAX(acc.max, d)
            return acc
        },
        reducer: (a1, a2) -> {
            a1.min = MIN(a1.min, a2.min)                a1.max = MAX(a1.max, a2.max)
            return a1
        })
    .map(t -> Row(t.carrier, t.res.min, t.res.max))

\begin{frame}{Conclusioni e Sviluppi Futuri}

    \textbf{Conclusioni}
    \begin{itemize}
        \item \textbf{Ecosistema Enterprise-Ready}: L'orchestrazione asincrona con Airflow combinata alla modularità di Spark e NiFi garantisce scalabilità e robustezza.
        \item \textbf{Ottimizzazione dell'I/O}: L'uso congiunto del formato colonnare Parquet e del Partition Pruning diminuisce l'overhead di lettura sul file system distribuito.
        \item \textbf{Analitica Avanzata e Sostenibile}: L'integrazione di Quantile Sketches e Model Selection automatizzato decodifica dinamiche complesse mantenendo costi computazionali accettabili.
    \end{itemize}

    \vspace{0.4cm}

    \textbf{Sviluppi Futuri}
    \begin{itemize}
        \item \textbf{Architettura Streaming}: Integrazione di Apache Kafka e Spark Structured Streaming per l'acquisizione della telemetria di volo in tempo reale.
        \item \textbf{Predictive Analytics on-the-fly}: Sfruttare i centroidi dei modelli ML pre-addestrati per classificare la stabilità della rete e prevedere anomalie in tempo reale.
    \end{itemize}
