\section{Extraction \& Ingestion Layer}

\begin{frame}[fragile]{NiFi Ingestion Pipeline: Approccio Data-Driven}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{itemize}
                \item \textbf{Design Data-Driven}: Pipeline stateless configurata dinamicamente tramite payload JSON iniettato all'avvio.

                \item \textbf{Ingestion \& Validazione}: \texttt{HandleHttpRequest} espone l'endpoint REST per l'avvio da Airflow, mentre \texttt{ValidateJson} esegue un controllo strutturale con schema rigoroso.

                \item \textbf{Propagazione del Dato}: \texttt{EvaluateJsonPath} mappa le chiavi JSON in Attributi del FlowFile, guidando in modo stateless il routing e le decisioni future tramite metadati nativi.

            \end{itemize}
        \end{column}

        \begin{column}{0.45\textwidth}
            \begin{block}{Payload JSON}
                \begin{semiverbatim}\tiny
\{
    "job_id": "FLIGHT_INGEST_run_123",
    "storage_raw": \{
        "type": "HDFS",
        "path": "/data/raw"
    \},
    "storage_preprocessed": \{
        "type": "S3",
        "bucket": "spark-flight-analysis",
        "path": "data/conv"
    \},
    "processing": \{
        "optimization_strategy": "PARTITION_..."
    \},
    "callback": \{
        "url": "http://airflow.../variables/v",
        "method": "PATCH",
        "headers": \{
            "Authorization": "Bearer eyJhbG..."
        \}
    \}
\}
                \end{semiverbatim}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Ingestion, Sicurezza e Gateway}
    \begin{itemize}
        \item \textbf{Estrazione}: Ricezione e spacchettamento degli archivi (\texttt{UnpackContent}).
        \item \textbf{Validazione SHA-1}:
        \begin{itemize}
            \item Controllo di integrità crittografica sui file estratti.
            \item Confronto hash atteso vs calcolato (\texttt{CryptographicHashContent}).
            \item Routing degli errori per isolare dati corrotti.
        \end{itemize}
        \item \textbf{Gateway \& Biforcazione}:
        \begin{itemize}
            \item \textbf{Ramo Raw}: Salvataggio as-is per garanzia del dato originale.
            \item \textbf{Ramo Preprocessing}: Inoltro dei record validati alla trasformazione.
        \end{itemize}
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Report/graphs/Nifi-Extract.png}
\end{frame}

\begin{frame}{Preprocessing e Ottimizzazione in Parquet}
    \begin{itemize}
        \item \textbf{Operatori NiFi per la Trasformazione}:
        \begin{itemize}
            \item \texttt{QueryRecord}/\texttt{UpdateRecord}: Filtraggio, casting e normalizzazione.
            \item \texttt{LookupRecord}: Arricchimento dei dati via file dizionario.
            \item \texttt{MergeRecord}: Prevenzione dello "Small Files Problem".
        \end{itemize}
        \item \textbf{Target Parquet}: Conversione con schema enforcement (Avro).
        \item \textbf{Strategie di Ottimizzazione}:
        \begin{itemize}
            \item \textbf{Partition Pruning}: Generazione dinamica dei path (\texttt{UpdateAttribute}).
            \item \textbf{Predicate Pushdown}: Ottimizzazione dei metadati per Spark.
        \end{itemize}
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Report/graphs/Nifi-Convert.png}
\end{frame}

\begin{frame}{Storage Finale e Handover ad Airflow}
    \begin{itemize}
        \item \textbf{Storage Distribuito}: Scrittura dei file Parquet ottimizzati in HDFS o S3.
        \item \textbf{Notifica di Completamento}:
        \begin{itemize}
            \item Chiamata REST via \texttt{InvokeHTTP} (PATCH/POST) alle API di Airflow.
            \item Handover del payload aggiornato con lo stato dell'elaborazione.
        \end{itemize}
        \item \textbf{Orchestrazione Analitica}: Sblocco automatico dei successivi job Spark in base alla strategia scelta.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Report/graphs/Nifi-Finalize.png}
\end{frame}
