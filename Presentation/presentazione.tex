\begin{frame}[fragile]{Query 1: Analisi Mensile Performance — RDD API}

    \vspace{-0.2cm}
    {\scriptsize \textbf{Obiettivo:} Aggregare i voli 2025 (AA vs DL) su base mensile. Calcolare min, max, media di \textit{DEP\_DELAY} (solo voli non cancellati) e il \textit{CANCELLATION RATE}.}
    
    \vspace{0.1cm}
    \hrule
    \vspace{0.1cm}

    \textbf{\footnotesize Paradigm 1: RDD API}

    \begin{alltt}\fontsize{6pt}{7pt}\selectfont
datasetVoli
    .mapToPair(v -> ([v.airline, v.month], v))
    .aggregateByKey(
        zeroValue: [sommaRitardi: 0, maxDelay: -\(\infty\), minDelay: +\(\infty\), nonCancellati: 0, totali: 0],
        combiner: (acc, v) -> \{
            acc.totali += 1
            if !v.cancelled then
                acc.sommaRitardi += v.dep_delay
                acc.maxDelay = MAX(acc.maxDelay, v.dep_delay)
                acc.minDelay = MIN(acc.minDelay, v.dep_delay)
                acc.nonCancellati += 1
            return acc
        \},
        reducer: (a1, a2) -> \{
            a1.sommaRitardi += a2.sommaRitardi
            a1.maxDelay = MAX(a1.maxDelay, a2.maxDelay)
            a1.minDelay = MIN(a1.minDelay, a2.minDelay)
            a1.nonCancellati += a2.nonCancellati
            a1.totali += a2.totali
            return a1
        \}
    )
    .map(tuple -> \{
        media = ROUND(tuple.sommaRitardi / tuple.nonCancellati, 2)
        tasso_canc = ROUND((tuple.totali - tuple.nonCancellati) / tuple.totali * 100, 2)
        return Row(tuple.month, tuple.airline, media, tuple.minDelay, tuple.maxDelay, tasso_canc)
    \})
    \end{alltt}

\end{frame}

    .mapToPair(f -> (f.getString(OP_UNIQUE_CARRIER), f))

sortedRDD = processedRDD.sortBy(r -> r.getDouble(2), false, partitions)

\begin{frame}{Query 3: Distribuzione e Percentili}
    \begin{itemize}
        \item \textbf{Obiettivo}: Statistiche orarie (Q1, Mediana, Q3).
        \item \textbf{RDD (La sfida)}: \texttt{groupByKey()} su milioni di record causerebbe OOM.
        \item \textbf{DataFrame / SQL (Approssimazione)}:
        \begin{itemize}
            \item \texttt{agg(expr("percentile\_approx(DEP\_DELAY, 0.5)"))}
            \item Uso dell'algoritmo nativo Greenwald-Khanna per stime efficienti.
        \end{itemize}
        \item \textbf{Soluzione RDD Custom}:
        \begin{itemize}
            \item \texttt{PercentileSketch sketch = PercentileSketch.from(config)}
            \item Integrazione di T-Digest e KLL per calcolo distribuito e mergeable.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Gestione Percentili in RDD: Strategie di Sketching}
    \begin{itemize}
        \item \textbf{Design Pattern Strategy}: Implementazione di algoritmi probabilistici in RDD.
        \item \textbf{T-Digest Strategy}:
        \begin{itemize}
            \item Struttura dati basata su centroidi pesati.
            \item Alta precisione nelle "code" della distribuzione (ritardi estremi).
        \end{itemize}
        \item \textbf{KLL Sketch Strategy}:
        \begin{itemize}
            \item Algoritmo stream-based con footprint di memoria fisso e predicibile.
            \item Merge delle strutture dati tra worker senza trasferimento di interi dataset.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Metodologia di Benchmarking}
    \begin{itemize}
        \item \textbf{Rigore Statistico}: Ogni combinazione (Query $\times$ Backend) è stata eseguita \textbf{5 volte} per garantire la stabilità delle misurazioni.
        \item \textbf{Telemetria}: Utilizzo del \texttt{JobTimerListener} (SparkListener) per isolare il \textit{tempo netto di calcolo} dall'overhead di sistema.
        \item \textbf{Hardware e Infrastruttura}:
        \begin{itemize}
            \item \textbf{EC2 (Self-Hosted)}: Master + Worker su istanze \texttt{t3.small}. Storage basato su HDFS.
            \item \textbf{EMR (Cloud-Native)}: 1 Master + 3 Core su istanze \texttt{m6g.xlarge} (ARM64). Architettura \textit{Shared-Nothing} su Amazon S3 via \texttt{s3a://}.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Confronto API: RDD vs DataFrame vs SQL}
    \begin{itemize}
        \item \textbf{DataFrame/SQL}: Massima stabilità e minori tempi di esecuzione nelle query di aggregazione (es. Q1) grazie all'ottimizzatore \textit{Catalyst} e alla gestione binaria off-heap di \textit{Tungsten}.
        \item \textbf{RDD}: Più performante in task atomici e logiche distribuite custom (Q2 e Q3), nonostante il maggior overhead di serializzazione JVM.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{../Results/charts/telemetry/ec2/part_prun/metrics_ec2_monthly_performance_part_prun_cropped.png}
\end{frame}

\begin{frame}{Ottimizzazione I/O: Pruning vs Pushdown}
    \begin{itemize}
        \item \textbf{Partition Pruning}: Strategia ottimale. Esclude i dati a livello di file system, garantendo un I/O bilanciato e l'assenza di \textit{Workload Skew} sui nodi.
        \item \textbf{Predicate Pushdown}: Richiede la scansione dei metadati Parquet. Genera un volume di lettura superiore e tende a sovraccaricare singoli nodi (colli di bottiglia in fase di scansione).
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.55\textheight,keepaspectratio]{../Results/charts/telemetry/ec2/pred_push/metrics_ec2_monthly_performance_pred_push_cropped.png}
\end{frame}

\begin{frame}{Overhead Infrastrutturale: EC2 vs EMR}
    \begin{itemize}
        \item \textbf{Standalone (EC2)}: Overhead di \textit{Wall Clock Time} significativo dovuto ai tempi di negoziazione delle risorse, bootstrap del cluster e latenze di rete HDFS.
        \item \textbf{Managed (EMR)}: L'integrazione nativa con S3 e la gestione elastica delle istanze contraggono sensibilmente i "tempi morti", massimizzando l'efficienza temporale del job.
    \end{itemize}
    \vfill

\end{frame}

\section{Risultati Analitici}

\begin{frame}{Query 3: Scelta dell'Euristica e Approssimazione}
    \begin{itemize}
        \item \textbf{Problema}: Calcolo dei percentili esatti su RDD (\texttt{groupByKey}) causa \textit{Out Of Memory} e colli di bottiglia distribuiti.
        \item \textbf{Soluzione}: Implementazione di algoritmi di \textbf{Quantile Sketching} (T-Digest e KLL).
        \item \textbf{Analisi Comparativa}:
        \begin{itemize}
            \item \textbf{KLL Sketch}: Elevata precisione teorica, ma footprint di memoria superiore.
            \item \textbf{T-Digest}: Ottimizzato per le "code" della distribuzione e più performante.
        \end{itemize}
        \item \textbf{Validazione}: Risultati pressoché identici sul dataset reale; la convergenza dei dati giustifica l'uso di algoritmi probabilistici.
        \item \textbf{Scelta Finale}: \textbf{T-Digest} per il miglior rapporto performance/accuratezza nel dominio aeronautico.
    \end{itemize}

    \centering
