\begin{frame}[fragile]{Query 1: Analisi Mensile Performance — RDD API}

    \vspace{-0.3cm}
    \begin{exampleblock}{\footnotesize Obiettivo}
        \scriptsize Aggregare i voli 2025 (AA vs DL) su base mensile. Calcolare min, max, media di \textit{DEP\_DELAY} (solo voli non cancellati) e il \textit{CANCELLATION RATE}.
    \end{exampleblock}
    
    \vspace{-0.2cm}

    % Usiamo il rddcodebox a tutta larghezza (senza colonne) visto che il codice è lungo
    \begin{rddcodebox}{Paradigm 1: Low-Level RDD API}
datasetVoli
    .mapToPair(v -> ([v.airline, v.month], v))
    .aggregateByKey(
        zeroValue: [sommaRitardi: 0, maxDelay: -inf, minDelay: +inf, nonCancellati: 0, totali: 0],
        combiner: (acc, v) -> {
            acc.totali += 1
            if !v.cancelled then
                acc.sommaRitardi += v.dep_delay
                acc.maxDelay = MAX(acc.maxDelay, v.dep_delay)
                acc.minDelay = MIN(acc.minDelay, v.dep_delay)
                acc.nonCancellati += 1
            return acc},
        reducer: (a1, a2) -> {
            a1.sommaRitardi += a2.sommaRitardi
            a1.maxDelay = MAX(a1.maxDelay, a2.maxDelay)
            a1.minDelay = MIN(a1.minDelay, a2.minDelay)
            a1.nonCancellati += a2.nonCancellati
            a1.totali += a2.totali
            return a1})
    .map(tuple -> {
        media = ROUND(tuple.sommaRitardi / tuple.nonCancellati, 2)
        tasso_canc = ROUND((tuple.totali - tuple.nonCancellati) / tuple.totali * 100, 2)
        return Row(tuple.month, tuple.airline, media, tuple.minDelay, tuple.maxDelay, tasso_canc)})
    \end{rddcodebox}

\end{frame}

\begin{frame}[fragile]{Query 2: Top 10 Ritardi e Scomposizione Cause — RDD API}

    \vspace{-0.3cm}
    \begin{exampleblock}{\footnotesize Obiettivo}
        \scriptsize Classifica prime 10 compagnie per \textit{ARR\_DELAY} medio decrescente (voli non cancellati/deviati con conteggio $\ge$ 500), mostrando l'incidenza media di ciascuna causa di ritardo.

    % Usiamo il rddcodebox a tutta larghezza visto il volume del codice
    \begin{rddcodebox}{Paradigm 1: Low-Level RDD API}
validFlights = flights.filter(f -> getSafe(f, CANCELLED) == 0 && getSafe(f, DIVERTED) == 0)
reducedRDD = validFlights
    .mapToPair(f -> (f.getString(OP_UNIQUE_CARRIER), f))
    .aggregateByKey(
        zeroValue: new double[7],  // [0: Count, 1: Arr, 2: Carrier, 3: Weather, 4: NAS, 5: Security, 6: Late]
        combiner: (acc, f) -> {
            acc[0] += 1.0;                       acc[1] += getSafe(f, ARR_DELAY);
            acc[2] += getSafe(f, CARRIER_DELAY);  acc[3] += getSafe(f, WEATHER_DELAY);
            acc[4] += getSafe(f, NAS_DELAY);      acc[5] += getSafe(f, SECURITY_DELAY);
            acc[6] += getSafe(f, LATE_AIRCRAFT_DELAY);
            return acc;},
        reducer: (acc1, acc2) -> {
            for (int i=0; i<7; i++) acc1[i] += acc2[i];
            return acc1;})
processedRDD = reducedRDD
    .filter(t -> t._2[0] >= 500.0)
    .map(t -> {
        double c = t._2[0];
        return RowFactory.create(t._1, (long)c, t._2[1]/c, t._2[2]/c, t._2[3]/c, t._2[4]/c, t._2[5]/c, t._2[6]/c);})
// Ordinamento globale ed estrazione Top 10 (Volume trascurabile post-filtraggio)
sortedRDD = processedRDD.sortBy(r -> r.getDouble(2), false, partitions)
top10RDD  = sortedRDD.zipWithIndex().filter(t -> t._2 < 10).map(t -> t._1).coalesce(1)
    \end{rddcodebox}

SELECT 
  OP_UNIQUE_CARRIER AS carrier, 
  COUNT(*)          AS num_flights, 
  ROUND(AVG(COALESCE(ARR_DELAY, 0)), 2) 
    AS arrdelay_mean, 
  ROUND(AVG(COALESCE(CARRIER_DELAY, 0)), 2) 
    AS carrier_delay_mean, 
  ROUND(AVG(COALESCE(WEATHER_DELAY, 0)), 2) 
    AS weather_delay_mean, 
  ROUND(AVG(COALESCE(NAS_DELAY, 0)), 2)

FROM flights 
WHERE CANCELLED = 0 AND DIVERTED = 0 
GROUP BY OP_UNIQUE_CARRIER 
HAVING num_flights >= 500 
ORDER BY arrdelay_mean DESC 
LIMIT 10

\begin{frame}[fragile]{Query 3: Percentili Orari e Min/Max Globali — RDD API}

    \vspace{-0.3cm}

// Subset condiviso e mantenuto in memoria per evitare scansioni multiple
validFlights = flights.filter(f -> f.getDouble(CANCELLED) == 0.0 && !f.isNullAt(DEP_DELAY)).cache()
// Pipeline 1: Calcolo Percentili Orari tramite Sketch distribuiti
PercentileSketch sketch = PercentileSketch.from(config.getPercentileAlgorithm())
hourlyRows = validFlights
    .filter(f -> !f.isNullAt(CRS_DEP_TIME))
    .mapToPair(f -> {
        int hour = (f.getInt(CRS_DEP_TIME) / 100) % 24;
        return new Tuple2<>(new Tuple2<>(f.getString(CARRIER), hour), f.getDouble(DEP_DELAY));})
    .combineByKey(sketch::init, sketch::update, sketch::merge)
    .map(t -> {
        double[] q = sketch.getQuantiles(t._2, 0.25, 0.50, 0.75, 0.90);
        return RowFactory.create(t._1._1, t._1._2, round2(q[0]), round2(q[1]), round2(q[2]), round2(q[3]));})
// Pipeline 2: Aggregazione Globale Min/Max per Compagnia
globalRows = validFlights
    .mapToPair(f -> new Tuple2<>(f.getString(CARRIER), f))
    .aggregateByKey(
        zeroValue: new double[]{Double.MAX_VALUE, -Double.MAX_VALUE},
        combiner: (acc, f) -> {
            double d = f.getDouble(DEP_DELAY);
            acc[0] = Math.min(acc[0], d); acc[1] = Math.max(acc[1], d);
            return acc;},
        reducer: (a1, a2) -> {
            a1[0] = Math.min(a1[0], a2[0]); a1[1] = Math.max(a1[1], a2[1]);
            return a1;})
    .map(t -> RowFactory.create(t._1, t._2[0], t._2[1]))
    \end{rddcodebox}

\begin{frame}{Monitoraggio e Raccolta Metriche di Performance}

    \begin{itemize}
        \item \textbf{Architettura Centralizzata (\texttt{PerformanceMetrics})}:
        \begin{itemize}
            \item \textbf{Design Pattern Singleton}: Istanza unica globalmente sincronizzata (\texttt{getInstance()}) per raccogliere metriche in modo thread-safe durante il ciclo di vita dell'app.

            \item \textbf{Dual-Clock Tracking}: Separazione netta tra il tempo reale percepito (\textit{Wall-Clock Time}) e il tempo computazionale effettivo speso sui nodi cluster (\textit{Spark Execution Time}).

        \vspace{0.4cm}

        \item \textbf{Disaccoppiamento via Event-Driven Listening (\texttt{JobTimerListener})}:
        \begin{itemize}
            \item \textbf{Estensione del \texttt{SparkListener}}: Intercettazione nativa e asincrona del ciclo di esecuzione di Spark senza sporcare la logica algoritmica delle query.

            \begin{itemize}
                \item \textbf{Job \& Stage Level}: Tracciamento di ID, durate pure e tempi morti spesi nel garbage collector JVM (\textit{GC Time}).
                \item \textbf{I/O \& Network Balance}: Misurazione al byte dei dati letti/scritti su file system e del carico di rete generato durante le fasi di \textit{Shuffle Read/Write}.
                \item \textbf{Worker Profiling (\texttt{onTaskEnd})}: Profilazione distribuita per singolo Executor (CPU time, task count, deserializzazione) per individuare colli di bottiglia o fenomeni di \textit{Data Skew}.

\begin{frame}{Metodologia di Benchmarking}
    \begin{itemize}
        \item \textbf{Rigore Statistico}: Ogni combinazione (Query $\times$ Backend) è stata eseguita \textbf{5 volte} per garantire la stabilità delle misurazioni.
        \item \textbf{Telemetria}: Utilizzo del \texttt{JobTimerListener} (SparkListener) per isolare il \textit{tempo netto di calcolo} dall'overhead di sistema.
        \item \textbf{Hardware e Infrastruttura}:
        \begin{itemize}
            \item \textbf{EC2 (Self-Hosted)}: Master + Worker su istanze \texttt{t3.small}. Storage basato su HDFS.
            \item \textbf{EMR (Cloud-Native)}: 1 Master + 3 Core su istanze \texttt{m6g.xlarge} (ARM64). Architettura \textit{Shared-Nothing} su Amazon S3 via \texttt{s3a://}.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Confronto API: RDD vs DataFrame vs SQL}
    \begin{itemize}
        \item \textbf{DataFrame/SQL}: Massima stabilità e minori tempi di esecuzione nelle query di aggregazione (es. Q1) grazie all'ottimizzatore \textit{Catalyst} e alla gestione binaria off-heap di \textit{Tungsten}.
        \item \textbf{RDD}: Più performante in task atomici e logiche distribuite custom (Q2 e Q3), nonostante il maggior overhead di serializzazione JVM.
    \end{itemize}
    \vfill

\end{frame}

\begin{frame}{Ottimizzazione I/O: Pruning vs Pushdown}
    \begin{itemize}
        \item \textbf{Partition Pruning}: Strategia ottimale. Esclude i dati a livello di file system, garantendo un I/O bilanciato e l'assenza di \textit{Workload Skew} sui nodi.
        \item \textbf{Predicate Pushdown}: Richiede la scansione dei metadati Parquet. Genera un volume di lettura superiore e tende a sovraccaricare singoli nodi (colli di bottiglia in fase di scansione).
    \end{itemize}

\begin{frame}{Overhead Infrastrutturale: EC2 vs EMR}
    \begin{itemize}
        \item \textbf{Standalone (EC2)}: Overhead di \textit{Wall Clock Time} significativo dovuto ai tempi di negoziazione delle risorse, bootstrap del cluster e latenze di rete HDFS.
        \item \textbf{Managed (EMR)}: L'integrazione nativa con S3 e la gestione elastica delle istanze contraggono sensibilmente i "tempi morti", massimizzando l'efficienza temporale del job.
    \end{itemize}

\section{Risultati Analitici}

\begin{frame}{Query 3: Scelta dell'Euristica e Approssimazione}
    \begin{itemize}
        \item \textbf{Problema}: Calcolo dei percentili esatti su RDD (\texttt{groupByKey}) causa \textit{Out Of Memory} e colli di bottiglia distribuiti.
        \item \textbf{Soluzione}: Implementazione di algoritmi di \textbf{Quantile Sketching} (T-Digest e KLL).
        \item \textbf{Analisi Comparativa}:
        \begin{itemize}
            \item \textbf{KLL Sketch}: Elevata precisione teorica, ma footprint di memoria superiore.
            \item \textbf{T-Digest}: Ottimizzato per le "code" della distribuzione e più performante.
        \end{itemize}
        \item \textbf{Validazione}: Risultati pressoché identici sul dataset reale; la convergenza dei dati giustifica l'uso di algoritmi probabilistici.
        \item \textbf{Scelta Finale}: \textbf{T-Digest} per il miglior rapporto performance/accuratezza nel dominio aeronautico.
    \end{itemize}
