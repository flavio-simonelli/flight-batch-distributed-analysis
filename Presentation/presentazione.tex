% Extraction & Ingestion Layer
\section{Extraction \& Ingestion Layer}

\begin{frame}{NiFi Ingestion Pipeline: Approccio Data-Driven}
    \begin{itemize}
        \item \textbf{Design Data-Driven}: Pipeline stateless configurata dinamicamente tramite payload JSON iniettato all'avvio (\texttt{EvaluateJsonPath}).
        \item \textbf{Parametri chiave del JSON}:
        \begin{itemize}
            \item \texttt{experiment.id}: Identificativo univoco del job.
            \item \texttt{ingestion}: Parametri sorgente (URL HTTP, SHA-1 atteso).
            \item \texttt{storage}: Percorsi dinamici per dati Raw e Preprocessed (HDFS/S3).
            \item \texttt{processing.strategy}: Scelta della strategia di ottimizzazione (\texttt{PARTITION PRUNING} o \texttt{PREDICATE PUSHDOWN}).
            \item \texttt{callback}: Configurazione per la notifica finale ad Airflow.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Ingestion, Sicurezza e Gateway}
    \begin{itemize}
        \item \textbf{Estrazione}: Ricezione e spacchettamento degli archivi (\texttt{UnpackContent}).
        \item \textbf{Validazione SHA-1}:
        \begin{itemize}
            \item Controllo di integrità crittografica sui file estratti.
            \item Confronto hash atteso vs calcolato (\texttt{CryptographicHashContent}).
            \item Routing degli errori per isolare dati corrotti.
        \end{itemize}
        \item \textbf{Gateway \& Biforcazione}:
        \begin{itemize}
            \item \textbf{Ramo Raw}: Salvataggio as-is per garanzia del dato originale.
            \item \textbf{Ramo Preprocessing}: Inoltro dei record validati alla trasformazione.
        \end{itemize}
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Report/graphs/Nifi-Extract.png}
\end{frame}

\begin{frame}{Preprocessing e Ottimizzazione in Parquet}
    \begin{itemize}
        \item \textbf{Operatori NiFi per la Trasformazione}:
        \begin{itemize}
            \item \texttt{QueryRecord}/\texttt{UpdateRecord}: Filtraggio, casting e normalizzazione.
            \item \texttt{LookupRecord}: Arricchimento dei dati via file dizionario.
            \item \texttt{MergeRecord}: Prevenzione dello "Small Files Problem".
        \end{itemize}
        \item \textbf{Target Parquet}: Conversione con schema enforcement (Avro).
        \item \textbf{Strategie di Ottimizzazione}:
        \begin{itemize}
            \item \textbf{Partition Pruning}: Generazione dinamica dei path (\texttt{UpdateAttribute}).
            \item \textbf{Predicate Pushdown}: Ottimizzazione dei metadati per Spark.
        \end{itemize}
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Report/graphs/Nifi-Convert.png}
\end{frame}

\begin{frame}{Storage Finale e Handover ad Airflow}
    \begin{itemize}
        \item \textbf{Storage Distribuito}: Scrittura dei file Parquet ottimizzati in HDFS o S3.
        \item \textbf{Notifica di Completamento}:
        \begin{itemize}
            \item Chiamata REST via \texttt{InvokeHTTP} (PATCH/POST) alle API di Airflow.
            \item Handover del payload aggiornato con lo stato dell'elaborazione.
        \end{itemize}
        \item \textbf{Orchestrazione Analitica}: Sblocco automatico dei successivi job Spark in base alla strategia scelta.
    \end{itemize}
    \vfill
    \centering
    \includegraphics[width=0.9\textwidth,height=0.45\textheight,keepaspectratio]{../Report/graphs/Nifi-Finalize.png}
\end{frame}

%Computation Layer
\section{Computation Layer}

\begin{frame}{Architettura e Astrazione: BaseQuery}
    \begin{itemize}
        \item \textbf{Ecosistema Spark}: Implementazione in Java utilizzando Spark Core (RDD), Spark SQL e MLlib.
        \item \textbf{Design Pattern Template Method}:
        \begin{itemize}
            \item Astrazione tramite la classe \texttt{BaseQuery.java}.
            \item Garanzia di coerenza tra i diversi paradigmi di esecuzione.
        \end{itemize}
        \item \textbf{Benchmarking Comparativo}: Ogni analisi implementa obbligatoriamente tre motori:
        \begin{itemize}
            \item \textbf{RDD API}: Controllo granulare su partizioni e serializzazione.
            \item \textbf{DataFrame API}: Astrazione strutturata (DSL) e ottimizzazione \textit{Catalyst}.
            \item \textbf{Spark SQL}: Interfaccia dichiarativa per logiche di business complesse.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Query 1: Analisi Mensile delle Performance}
    \begin{itemize}
        \item \textbf{Obiettivo}: Ritardi medi, estremi e tasso di cancellazione mensile per vettore.
        \item \textbf{RDD (\texttt{aggregateByKey})}: 
        \begin{itemize}
            \item \texttt{pairs.aggregateByKey(zeroValue, seqOp, combOp)}
            \item Aggregazione locale su partizione per minimizzare il traffico di rete.
        \end{itemize}
        \item \textbf{DataFrame (DSL)}:
        \begin{itemize}
            \item \texttt{flights.groupBy("MONTH", "CARRIER").agg(avg("DELAY"), ...)}
        \end{itemize}
        \item \textbf{Spark SQL}:
        \begin{itemize}
            \item \texttt{SELECT MONTH, AVG(DELAY) FROM flights GROUP BY MONTH, ...}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Query 2: Ranking dei Ritardi in Arrivo}
    \begin{itemize}
        \item \textbf{Obiettivo}: Identificazione delle Top-K tratte più critiche.
        \item \textbf{RDD (Manuale)}:
        \begin{itemize}
            \item \texttt{mapToPair().reduceByKey().takeOrdered(10, comparator)}
        \end{itemize}
        \item \textbf{DataFrame (Window Functions)}:
        \begin{itemize}
            \item \texttt{agg(avg("ARR\_DELAY")).orderBy(col("mean").desc()).limit(10)}
        \end{itemize}
        \item \textbf{Spark SQL}:
        \begin{itemize}
            \item \texttt{SELECT carrier, AVG(ARR\_DELAY) as mean FROM flights ... ORDER BY mean DESC LIMIT 10}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Query 3: Distribuzione e Percentili}
    \begin{itemize}
        \item \textbf{Obiettivo}: Statistiche orarie (Q1, Mediana, Q3).
        \item \textbf{RDD (La sfida)}: \texttt{groupByKey()} su milioni di record causerebbe OOM.
        \item \textbf{DataFrame / SQL (Approssimazione)}:
        \begin{itemize}
            \item \texttt{agg(expr("percentile\_approx(DEP\_DELAY, 0.5)"))}
            \item Uso dell'algoritmo nativo Greenwald-Khanna per stime efficienti.
        \end{itemize}
        \item \textbf{Soluzione RDD Custom}:
        \begin{itemize}
            \item \texttt{PercentileSketch sketch = PercentileSketch.from(config)}
            \item Integrazione di T-Digest e KLL per calcolo distribuito e mergeable.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Gestione Percentili in RDD: Strategie di Sketching}
    \begin{itemize}
        \item \textbf{Design Pattern Strategy}: Implementazione di algoritmi probabilistici in RDD.
        \item \textbf{T-Digest Strategy}:
        \begin{itemize}
            \item Struttura dati basata su centroidi pesati.
            \item Alta precisione nelle "code" della distribuzione (ritardi estremi).
        \end{itemize}
        \item \textbf{KLL Sketch Strategy}:
        \begin{itemize}
            \item Algoritmo stream-based con footprint di memoria fisso e predicibile.
            \item Merge delle strutture dati tra worker senza trasferimento di interi dataset.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Machine Learning: Airline Clustering}
    \begin{itemize}
        \item \textbf{Obiettivo}: Raggruppamento di vettori aerei con profili operativi simili.
        \item \textbf{Modello K-Means} (\textit{Spark MLlib}):
        \begin{itemize}
            \item Inizializzazione tramite \texttt{k-means||} (variante parallela di k-means++).
            \item Convergenza distribuita sui centroidi dei cluster.
        \end{itemize}
        \item \textbf{Data Pipeline}:
        \begin{itemize}
            \item \textbf{VectorAssembler}: Consolidamento delle features analitiche.
            \item \textbf{StandardScaler}: Normalizzazione fondamentale per la distanza Euclidea.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Telemetria Netta: Custom Spark Listener}
    \begin{itemize}
        \item \textbf{Obiettivo}: Misurazione rigorosa dei tempi di esecuzione per il benchmarking.
        \item \textbf{JobTimerListener}:
        \begin{itemize}
            \item Estensione di \texttt{SparkListener} per intercettare gli eventi del DAG Scheduler.
            \item Monitoraggio di \texttt{onJobStart} e \texttt{onJobEnd}.
        \end{itemize}
        \item \textbf{Precisione}: Eliminazione degli overhead di sistema, I/O Driver e task "lazy", garantendo metriche focalizzate puramente sul tempo di calcolo distribuito.
    \end{itemize}
\end{frame}
